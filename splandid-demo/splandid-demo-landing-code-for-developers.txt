          const activityHash = root.querySelector("#apActivityHash")?.value.trim();
          const phone = root.querySelector("#apPhoneNumber")?.value.trim();

          if (!activityHash || !phone) {
            setMessage("Vul eerst de activity hash en het telefoonnummer in.", "is-error");
            return;
          }

          if (!phone.startsWith("+")) {
            setMessage("Gebruik het internationale telefoonformaat, bijvoorbeeld +31612345678.", "is-error");
            return;
          }

          try {
            if (submit) {
              submit.disabled = true;
              submit.textContent = "Call wordt gestart...";
            }
            setStep(3);
            setMessage("Webhook wordt aangeroepen. Als de activiteit goed klaarstaat, gaat de telefoon binnen ongeveer 15 seconden.", "");

            const response = await fetch(webhookUrl, {
              method: "POST",
              headers: { "Content-Type": "application/json" },
              body: JSON.stringify({ activity_hash: activityHash, phone })
            });

            if (!response.ok) throw new Error("Webhook returned " + response.status);

            setMessage("Call gestart. Neem op, spreek een slot af met Alex en refresh daarna dezelfde activiteit in Splandid.", "is-success");
            window.setTimeout(() => setStep(4), 1200);
          } catch (error) {
            setStep(2);
            setMessage("De call kon niet worden gestart. Controleer de hash, het telefoonnummer en probeer opnieuw.", "is-error");
          } finally {
            if (submit) {
              submit.disabled = false;
              submit.textContent = "Plan deze activiteit";
            }
          }
        });
      }

      root.querySelectorAll('a[href^="#"]').forEach(link => {
        link.addEventListener("click", event => {
          const targetId = link.getAttribute("href").slice(1);
          const target = document.getElementById(targetId);
          if (!target) return;
          event.preventDefault();
          target.scrollIntoView({ behavior: "smooth", block: "start" });
          if (window.history && window.history.replaceState) {
            window.history.replaceState(null, "", "#" + targetId);
          }
        });
      });

      setStep(1);
    })();
  </script>
</div>
